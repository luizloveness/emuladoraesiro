// FUNÇÃO DE SLOTAGEM
function	script	Func_Socket2	{
	.@item_id = getarg(0);
	.@new_item = getarg(1);
	.@min_rand = getarg(2);
	.@max_rand = getarg(3);
	.@zeny_req = getarg(4);
	.@item_req1 = getarg(5);
	.@req_amt1 = getarg(6);
	.@item_req2 = getarg(7,0);
	.@req_amt2 = getarg(8,0);

	disable_items;
	mes "[Leablem]";
	mes "Bring me:";
	mes "- " + (.@zeny_req * 1000) + " zeny";
	mes "- " + .@req_amt1 + "x " + getitemname(.@item_req1);
	if (.@item_req2) mes "- " + .@req_amt2 + "x " + getitemname(.@item_req2);
	mes "- " + getitemname(.@item_id);
	next;

	if ((Zeny >= .@zeny_req*1000) &&
	    (countitem(.@item_req1) >= .@req_amt1) &&
	    (.@item_req2 == 0 || countitem(.@item_req2) >= .@req_amt2) &&
	    countitem(.@item_id) > 0) {

		set .@a, rand(1,100);
		if (.@a > .@min_rand && .@a < .@max_rand) {
			specialeffect EF_LORD;
			delitem .@item_id,1;
			delitem .@item_req1,.@req_amt1;
			if (.@item_req2) delitem .@item_req2,.@req_amt2;
			Zeny -= .@zeny_req*1000;
			getitem .@new_item,1;
			mes "Success!";
		} else {
			specialeffect EF_SUI_EXPLOSION;
			delitem .@item_id,1;
			delitem .@item_req1,.@req_amt1;
			if (.@item_req2) delitem .@item_req2,.@req_amt2;
			Zeny -= .@zeny_req*1000;
			mes "Failed...";
		}
	} else {
		mes "You don't have the required items.";
	}
	close;
}
