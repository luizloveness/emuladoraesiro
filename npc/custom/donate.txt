	function ClaimCredits {
		.@show_messages = getarg(0);

			"WHERE `account_id` = " + .@account_id + " AND `status` > 0 AND `pending` > 0 ORDER BY `id` ASC",
			.@id, .@value, .@pending, .@state$, .@reference$, .@status, .@coupon$
		);

		.@size = getarraysize(.@id);
		if (!.@size) {
			if (.@show_messages) {
				mes "Você não possui créditos pendentes para resgatar no momento.";
				end;
			}
			return;
		}

		.@sender$ = "Administração";
		.@title$ = "Obrigado por sua doação";

		for (.@j = 0; .@j < .@size; .@j++) {
			.@total = .@pending[.@j];
			.@bonus = 0;

			for (.@i = 0; .@i < getarraysize(.streamers$); .@i += 3) {
				if (.@coupon$[.@j] == .streamers$[.@i+1]) {
					.@bonus += (.@pending[.@j] * .streamers_bonus) / 100;
					break;
				}
}


			if (.have_bonus) {
				for (.@i = 0; .@i < getarraysize(.bonus); .@i += 2) {
					if (.@value[.@j] >= .bonus[.@i])
						.@bonus += (.@pending[.@j] * .bonus[.@i+1]) / 100;
				}
			}

			.@total += .@bonus;

			if (.@show_messages) {
				.@body$ = "Olá, " + strcharinfo(0) + "!\r\n"
						+ "Nós, da equipe " + .server$ + ", agradecemos de coração pela sua doação.\r\n"
						+ "Seu apoio é essencial para mantermos o servidor ativo e sempre melhorando. Muito obrigado por fazer parte dessa jornada conosco!\r\n"
						+ "Seus créditos foram creditados com sucesso e já estão disponíveis em sua conta.\r\n\r\n"
						+ "ID da Transação: " + .@reference$[.@j] + "\r\n"
						+ "Valor da Transação: R$" + .@value[.@j] + ",00\r\n";

				if (.@coupon$[.@j] != "")
					.@body$ += "Cupom utilizado: " + .@coupon$[.@j] + "\r\n";

				if (.@bonus > 0)
					.@body$ += "Bônus recebido: " + .@bonus + " créditos\r\n";

				.@body$ += "Total creditado: " + .@total + " créditos\r\n";

				mail .@char_id, .@sender$, .@title$, .@body$;
			}

			set #CASHPOINTS, #CASHPOINTS + .@total;
			query_sql "UPDATE `donate_system` SET `pending` = 0, `unique_id` = '" + .@gepard + "' WHERE `id` = " + .@id[.@j];
		}

		if (.@show_messages) {
			mes "Seus créditos foram resgatados com sucesso!";
			end;
		}
	}

	// Função responsável por gerar o código aleatório
	function uCode {
		.@s = getarg(0, 5); 
		.@rc$ = "";

		.@code_size = getarraysize(.characters_code$);

		for (.@i = 0; .@i < .@s; .@i++)
			.@rc$ += .characters_code$[rand(.@code_size)];

		.@tick = gettimetick(2);
		.@tick$ = "" + .@tick;

		while (getstrlen(.@tick$) < 10)
			.@tick$ = "0" + .@tick$;

		.@rc$ += substr(.@tick$, 5, 9);

		while (getstrlen(.@rc$) < 11)
			.@rc$ += .characters_code$[rand(.@code_size)];

		return .@rc$;
	}

function	script	donate	{

    set .@int,  getarg(0);
    set .@sep$, getarg(1,",");
    set .@str$, "";
    if ( .@int < 0 ) {
    set .@int, -.@int;
    set .@neg, 1;
}
    set .@len, getstrlen(.@int+"") - .@neg;
    for ( set .@i,0; .@i<.@len; set .@i,.@i+1 )
    set .@str$, ( .@i % 3 == 2 && .@i < .@len-1 ? .@sep$ : "" ) + ( .@int / pow(10,.@i) ) % 10 + .@str$;
    return ( .@neg ? "-" : "" ) + .@str$;
}
