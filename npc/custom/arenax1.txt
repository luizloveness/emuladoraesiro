				"(season,class_id,char_id,account_id,name,points,item_id,duration_days,status,rewarded_at) "+
				"VALUES ("+
				"'"+.@season$+"',"+
				.@reward_class_id+","+
				.@winner_id+","+
				.@winner_account_id+",'"+
				escape_sql(.@winner_name$)+"',"+
				.@winner_points+","+
				"31818,"+
				"29,"+
				"0,"+
				"NOW())"
			);

			// ------------------------------------------------
			// CONFIRMAR REGISTRO
			// ------------------------------------------------

			.@reward_id = 0;

			query_sql(
				"SELECT reward_id "+
				"FROM x1_arena_rewards "+
				"WHERE season = '"+.@season$+"' "+
				"AND class_id = "+.@reward_class_id+
				" AND char_id = "+.@winner_id+
				" AND status = 0 "+
				"ORDER BY reward_id DESC "+
				"LIMIT 1",
				.@reward_id
			);

				continue;

			// ------------------------------------------------
			// CONTABILIZAR
			// ------------------------------------------------

			.@premiados++;

			// ------------------------------------------------
			// ANÚNCIO
			// ------------------------------------------------

				.@reward_class_name$+

				.@winner_points+
				" pontos! "+
				"O Costume Top Player será entregue "+
				"automaticamente no próximo login.",
				bc_all;
		}

		// ====================================================
		// RESULTADO
		// ====================================================

		mes "[Arena 1vs1]";
		mes "^00AAFFPREMIAÇÃO REGISTRADA!^000000";
		mes "";
		mes "Temporada: ^00AAFF"+
			.@season$+
			"^000000";
		mes "";
		mes "Prêmios registrados: ^00AAFF"+
			.@premiados+
			"^000000";
		mes "Classes sem jogadores: ^777777"+
			.@sem_jogador+
			"^000000";
		mes "Já processados: ^777777"+
			.@ja_premiados+
			"^000000";
		mes "";
		mes "Item: ^00AAFF31818^000000";
		mes "Entrega: ^00AAFFNo próximo login^000000";
		mes "Duração: ^00AAFF29 dias^000000";
		mes "";
		mes "^777777O item não é entregue neste momento.^000000";
		mes "^777777A duração começa somente no login.^000000";

		close;
	}

	// SAIR
	// ========================================================

	close;
}

// SCRIPT INVISÍVEL - MORTE
// ============================================================

-	script	X1_Arena_Death	-1,{

OnPCDieEvent:

	.@char_id = getcharid(0);
	.@account_id = getcharid(3);
	.@name$ = strcharinfo(0);

	if (.@char_id <= 0)
		end;

	// ========================================================
	// LOCALIZAR PARTIDA ATIVA
	// ========================================================

	.@match_id = 0;
	.@char_id_1 = 0;
	.@char_id_2 = 0;
	.@class_id = 0;

	query_sql(
		"SELECT match_id,char_id_1,char_id_2,class_id "+
		"FROM x1_arena_matches "+
		"WHERE status = 'active' "+
		"AND (char_id_1 = "+.@char_id+
		" OR char_id_2 = "+.@char_id+") "+
		"LIMIT 1",
		.@match_id,
		.@char_id_1,
		.@char_id_2,
		.@class_id
	);

	if (.@match_id <= 0)
		end;

	// ========================================================
	// IDENTIFICAR VENCEDOR
	// ========================================================

	if (.@char_id == .@char_id_1) {

		.@winner_id = .@char_id_2;
		.@loser_id = .@char_id;

	} else {

		.@winner_id = .@char_id_1;
		.@loser_id = .@char_id;
	}

	// ========================================================
	// PROTEÇÃO CONTRA DUPLA MORTE
	// ========================================================

	query_sql(
		"UPDATE x1_arena_matches "+
		"SET status = 'finished', "+
		"winner_id = "+.@winner_id+", "+
		"loser_id = "+.@loser_id+", "+
		"finished_at = NOW() "+
		"WHERE match_id = "+.@match_id+
		" AND status = 'active'"
	);

	// ========================================================
	// CONFIRMAR PROCESSAMENTO
	// ========================================================

	.@processed = 0;

	query_sql(
		"SELECT COUNT(*) "+
		"FROM x1_arena_matches "+
		"WHERE match_id = "+.@match_id+
		" AND status = 'finished' "+
		" AND loser_id = "+.@loser_id,
		.@processed
	);

	if (.@processed <= 0)
		end;

	// ========================================================
	// DADOS DO VENCEDOR
	// ========================================================

	.@winner_account_id = 0;
	.@winner_name$ = "";
	.@winner_class_id = 0;

	query_sql(
		"SELECT account_id,name,class "+
		"FROM `char` "+
		"WHERE char_id = "+.@winner_id+
		" LIMIT 1",
		.@winner_account_id,
		.@winner_name$,
		.@winner_class_id
	);

	if (.@class_id <= 0)
		.@class_id = .@winner_class_id;

	.@season$ = gettimestr("%Y%m",21);

	// ========================================================
	// RANKING VENCEDOR
	// ========================================================

	.@winner_exists = 0;

	query_sql(
		"SELECT COUNT(*) "+
		"FROM x1_arena_ranking "+
		"WHERE char_id = "+.@winner_id+
		" AND season = '"+.@season$+"'",
		.@winner_exists
	);

	if (.@winner_exists > 0) {

		query_sql(
			"UPDATE x1_arena_ranking "+
			"SET points = points + 2, "+
			"name = '"+escape_sql(.@winner_name$)+"', "+
			"account_id = "+.@winner_account_id+", "+
			"class_id = "+.@class_id+
			" WHERE char_id = "+.@winner_id+
			" AND season = '"+.@season$+"'"
		);

	} else {

		query_sql(
			"INSERT INTO x1_arena_ranking "+
			"(account_id,char_id,name,class_id,points,season) "+
			"VALUES ("+
			.@winner_account_id+","+
			.@winner_id+",'"+
			escape_sql(.@winner_name$)+"',"+
			.@class_id+
			",2,'"+
			.@season$+"')"
		);
	}

	// ========================================================
	// DADOS DO DERROTADO
	// ========================================================

	.@loser_account_id = 0;
	.@loser_name$ = "";
	.@loser_class_id = 0;

	query_sql(
		"SELECT account_id,name,class "+
		"FROM `char` "+
		"WHERE char_id = "+.@loser_id+
		" LIMIT 1",
		.@loser_account_id,
		.@loser_name$,
		.@loser_class_id
	);

	// ========================================================
	// RANKING DERROTADO
	// ========================================================

	.@loser_exists = 0;

	query_sql(
		"SELECT COUNT(*) "+
		"FROM x1_arena_ranking "+
		"WHERE char_id = "+.@loser_id+
		" AND season = '"+.@season$+"'",
		.@loser_exists
	);

	if (.@loser_exists > 0) {

		query_sql(
			"UPDATE x1_arena_ranking "+
			"SET points = GREATEST(points - 1,0), "+
			"name = '"+escape_sql(.@loser_name$)+"', "+
			"account_id = "+.@loser_account_id+", "+
			"class_id = "+.@class_id+
			" WHERE char_id = "+.@loser_id+
			" AND season = '"+.@season$+"'"
		);

	} else {

		query_sql(
			"INSERT INTO x1_arena_ranking "+
			"(account_id,char_id,name,class_id,points,season) "+
			"VALUES ("+
			.@loser_account_id+","+
			.@loser_id+",'"+
			escape_sql(.@loser_name$)+"',"+
			.@class_id+
			",0,'"+
			.@season$+"')"
		);
	}

	// ========================================================
	// ANÚNCIO
	// ========================================================

	announce
		"[Arena 1vs1] "+
		.@loser_name$+
		" foi derrotado na Arena 1vs1!",
		bc_all;

	// ========================================================
	// DERROTADO
	// ========================================================

	dispbottom
		"[Arena 1vs1] Você perdeu a batalha!";

	dispbottom
		"[Arena 1vs1] -1 ponto no ranking.";

	dispbottom
		"[Arena 1vs1] Você será retirado da Arena.";

	warp
		"prontera",
		155,
		180;

	// ========================================================
	// ESPERAR VENCEDOR
	// ========================================================

	sleep 5000;

	// ========================================================
	// CONECTAR VENCEDOR
	// ========================================================

	if (.@winner_account_id <= 0)
		end;

	if (!attachrid(.@winner_account_id))
		end;

	// --------------------------------------------------------
	// SEGURANÇA
	// --------------------------------------------------------

	if (strcharinfo(3) != "force_3-3")
		end;

	// ========================================================
	// VENCEDOR
	// ========================================================

	dispbottom
		"[Arena 1vs1] Você venceu a batalha!";

	dispbottom
		"[Arena 1vs1] +2 pontos no ranking.";

	dispbottom
		"[Arena 1vs1] Retornando para Prontera...";

	warp
		"prontera",
		155,
		180;

	end;
}

// ============================================================
// VERIFICAÇÃO E ENTREGA DA PREMIAÇÃO NO LOGIN
// ============================================================
//
// O prêmio NÃO é entregue quando o GM realiza a premiação.
//
// Registro:
//
// status = 0 -> prêmio pendente
//
// No login do campeão:
//
// 1. Localiza a premiação pelo char_id.
// 2. Verifica o peso.
// 3. Entrega o item 31818.
// 4. Altera status para 1.
// 5. Os 29 dias começam neste momento.
//
// ============================================================

-	script	X1_Arena_Reward_Check	-1,{

OnPCLoginEvent:

	.@char_id = getcharid(0);

	if (.@char_id <= 0)
		end;

	// ========================================================
	// LOCALIZAR PREMIAÇÃO PENDENTE
	// ========================================================

	.@reward_id = 0;
	.@class_id = 0;
	.@item_id = 0;
	.@duration_days = 0;
	.@points = 0;
	.@season$ = "";

	query_sql(
		"SELECT reward_id,class_id,item_id,duration_days,points,season "+
		"FROM x1_arena_rewards "+
		"WHERE char_id = "+.@char_id+
		" AND status = 0 "+
		"ORDER BY reward_id ASC "+
		"LIMIT 1",
		.@reward_id,
		.@class_id,
		.@item_id,
		.@duration_days,
		.@points,
		.@season$
	);

	// ========================================================
	// NÃO POSSUI PREMIAÇÃO
	// ========================================================

	if (.@reward_id <= 0)
		end;

	// ========================================================
	// GARANTIR ITEM VÁLIDO
	// ========================================================

	if (.@item_id <= 0)
		.@item_id = 31818;

	if (.@duration_days <= 0)
		.@duration_days = 29;

	// ========================================================
	// VERIFICAR PESO
	// ========================================================

	if (!checkweight(.@item_id,1)) {

		dispbottom
			"[Arena 1vs1] Você possui uma premiação mensal pendente.";

		dispbottom
			"[Arena 1vs1] Libere espaço/peso no inventário e reconecte.";

		end;
	}

	// ========================================================
	// NOME DA CLASSE
	// ========================================================

	.@class_name$ = "";

	switch (.@class_id) {

		case 4001:
			.@class_name$ = "Novice High";
			break;

		case 4008:
			.@class_name$ = "Lord Knight";
			break;

		case 4009:
			.@class_name$ = "High Priest";
			break;

		case 4010:
			.@class_name$ = "High Wizard";
			break;

		case 4011:
			.@class_name$ = "Whitesmith";
			break;

		case 4012:
			.@class_name$ = "Sniper";
			break;

		case 4013:
			.@class_name$ = "Assassin Cross";
			break;

		case 4015:
			.@class_name$ = "Paladin";
			break;

		case 4016:
			.@class_name$ = "Champion";
			break;

		case 4017:
			.@class_name$ = "Professor";
			break;

		case 4018:
			.@class_name$ = "Stalker";
			break;

		case 4019:
			.@class_name$ = "Creator";
			break;

		case 4020:
			.@class_name$ = "Clown";
			break;

		case 4021:
			.@class_name$ = "Gypsy";
			break;

		default:
			.@class_name$ = "Classe desconhecida";
			break;
	}

	// ========================================================
	// ENTREGA
	// ========================================================

	.@duration_seconds = .@duration_days * 86400;

	rentitem
		.@item_id,
		.@duration_seconds;

	// ========================================================
	// CONFIRMAR ENTREGA
	// ========================================================

	query_sql(
		"UPDATE x1_arena_rewards "+
		"SET status = 1, rewarded_at = NOW() "+
		"WHERE reward_id = "+.@reward_id+
		" AND char_id = "+.@char_id+
		" AND status = 0"
	);

	// ========================================================
	// MENSAGENS PARA O CAMPEÃO
	// ========================================================

	dispbottom
		"================================================";

	dispbottom
		"[Arena 1vs1] PARABÉNS!";

	dispbottom
		"[Arena 1vs1] Você foi o campeão mensal da classe "+
		.@class_name$+"!";

	dispbottom
		"[Arena 1vs1] Você recebeu o Costume Top Player.";

	dispbottom
		"[Arena 1vs1] Item: 31818";

	dispbottom
		"[Arena 1vs1] Duração: "+
		.@duration_days+
		" dias.";

	dispbottom
		"[Arena 1vs1] Temporada: "+
		.@season$;

	dispbottom
		"================================================";

	// ========================================================
	// ANÚNCIO GLOBAL
	// ========================================================

	.@winner_name$ = strcharinfo(0);

	announce
		"[Arena 1vs1] CAMPEÃO "+
		.@class_name$+
		": "+
		.@winner_name$+
		" entrou no servidor e recebeu o Costume Top Player por "+
		.@duration_days+
		" dias!",
		bc_all;

	end;
}
