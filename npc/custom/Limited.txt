	// ============================================================
	// ADICIONA MAPA + ITENS
	// ============================================================

function	func_Add	{

	.@gmlv = getarg(0);
	.@map$ = getarg(1);

	// Salva nível de GM permitido
	setd(
		".li_gmlv_"+.@map$,
		.@gmlv
	);

	// Quantidade de posições utilizadas no array
	.@size = 0;

	// Argumentos:
	// 0 = GM Level
	// 1 = mapa
	// 2 = item
	// 3 = quantidade
	// 4 = item
	// 5 = quantidade
	// ...

	.@i = 2;

	while (.@i + 1 < getargcount()) {

		.@itemid = getarg(.@i);
		.@max = getarg(.@i + 1);

		// Segurança
		if (.@itemid > 0) {

			setd(
				".li_item_"+.@map$+"["+.@size+"]",
				.@itemid
			);

			setd(
				".li_item_"+.@map$+"[" + (.@size + 1) + "]",
				.@max
			);

			.@size += 2;
		}

		.@i += 2;
	}

	setd(
		".li_item_size_"+.@map$,
		.@size
	);

	// Ativa OnPCLoadMapEvent
	setmapflag(
		.@map$,
		mf_loadevent
	);

	return;
}


	// ============================================================
	// VERIFICA ITENS DO JOGADOR
	// ============================================================
	//
	// Retorno:
	// 0 = permitido
	// 1 = proibido
	//
	// A função verifica:
	//
	// 1. Random Options
	// 2. Quantidade máxima dos itens proibidos
	// 3. Itens equipados
	//
	// ============================================================

function	func_CheckMapItems	{

	.@map$ = getarg(0);

	.@size = getd(
		".li_item_size_"+.@map$
	);

	if (.@size <= 0)
		return 0;


	// ========================================================
	// CARREGA INVENTÁRIO
	// ========================================================

	getinventorylist;


	// ========================================================
	// VERIFICA RANDOM OPTIONS
	// ========================================================

	for (
		.@i = 0;
		.@i < @inventorylist_count;
		.@i++
	) {

		.@itemid = @inventorylist_id[.@i];

		if (.@itemid <= 0)
			continue;

		if (
			@inventorylist_option_id1[.@i] > 0 ||
			@inventorylist_option_id2[.@i] > 0 ||
			@inventorylist_option_id3[.@i] > 0 ||
			@inventorylist_option_id4[.@i] > 0 ||
			@inventorylist_option_id5[.@i] > 0
		) {

			mes "[WoE]";
			mes "^FF0000Random Option detectada!^000000";
			mes " ";
			mes "O item:";
			mes "^FFD700"+getitemname(.@itemid)+"^000000";
			mes " ";
			mes "possui Random Option.";
			mes "Este tipo de item não é permitido neste mapa.";

			return 1;
		}
	}


	// ========================================================
	// VERIFICA ITENS PROIBIDOS
	// ========================================================
	//
	// Não usamos copyarray aqui.
	// Lemos diretamente os valores salvos.
	//
	// [0] = ID
	// [1] = quantidade
	// [2] = ID
	// [3] = quantidade
	// ...
	// ========================================================

	.@i = 0;
	.@fail = 0;

	while (.@i < .@size) {

		.@itemid = getd(
			".li_item_"+.@map$+"["+.@i+"]"
		);

		.@max = getd(
			".li_item_"+.@map$+"[" + (.@i + 1) + "]"
		);


		// ====================================================
		// PROTEÇÃO CONTRA ITEM 0
		// ====================================================

		if (.@itemid <= 0) {

			.@i += 2;
			continue;
		}


		// ====================================================
		// QUANTIDADE NO INVENTÁRIO
		// ====================================================

		.@countitem = countitem(.@itemid);


		// ====================================================
		// QUANTIDADE EQUIPADA
		// ====================================================

		.@isequippedcnt = isequippedcnt(.@itemid);


		// ====================================================
		// COMPARAÇÃO
		// ====================================================

		if (
			(.@countitem + .@isequippedcnt) > .@max
		) {

			mes "[WoE]";
			mes "^FF0000Item proibido detectado!^000000";
			mes " ";
			mes "^FFD700"+getitemname(.@itemid)+"^000000";
			mes " ";
			mes "Quantidade permitida: ^00FF00"+.@max+"^000000";
			mes "Quantidade encontrada: ^FF0000"+
				(.@countitem + .@isequippedcnt)+
				"^000000";

			.@fail++;
		}

		.@i += 2;
	}


	// ========================================================
	// RESULTADO
	// ========================================================

	if (.@fail > 0) {

		mes " ";
		mes "^FF0000Remova os itens proibidos^000000";
		mes "para permanecer neste mapa.";

		return 1;
	}

	return 0;
}

	// ============================================================
	// ENTRADA NO MAPA
	// ============================================================

	OnPCLoadMapEvent:

		.@map$ = strcharinfo(3);

		// Não é mapa controlado
		if (
			getd(".li_item_size_"+.@map$) <= 0
		)
			end;


		// ========================================================
		// GM BYPASS
		// ========================================================

		if (
			getgmlevel() >=
			getd(".li_gmlv_"+.@map$)
		)
			end;


		// ========================================================
		// CONTROLE DO TIMER
		// ========================================================
		//
		// Evita criar vários timers simultâneos
		// para o mesmo personagem.
		//
		// ========================================================

		.@timer_active = #limit_item_timer;

		#limit_item_timer = 1;
		#limit_item_check_count = 0;


		// ========================================================
		// VERIFICAÇÃO IMEDIATA
		// ========================================================

		if (
			func_CheckMapItems(.@map$)
		) {

			close2;

			#limit_item_timer = 0;
			#limit_item_check_count = 0;

			warp "SavePoint",0,0;

			end;
		}


		// ========================================================
		// PRIMEIRO TIMER
		// ========================================================
		//
		// Só cria um novo timer se ainda não existir
		// um timer ativo para este personagem.
		//
		// ========================================================

		if (!.@timer_active) {

			addtimer 200,
				.npc_name$+"::OnCheck";
		}

		end;


	// ============================================================
	// TIMER DE VERIFICAÇÃO
	// ============================================================

OnCheck:

	// Timer não está ativo
	if (!#limit_item_timer)
		end;


	// ========================================================
	// MAPA ATUAL
	// ========================================================

	.@map$ = strcharinfo(3);


	// ========================================================
	// SAIU DO MAPA RESTRITO
	// ========================================================

	.@size = getd(
		".li_item_size_"+.@map$
	);

	if (.@size <= 0) {

		#limit_item_timer = 0;
		#limit_item_check_count = 0;

		end;
	}


	// ========================================================
	// GM BYPASS
	// ========================================================

	if (
		getgmlevel() >=
		getd(".li_gmlv_"+.@map$)
	) {

		#limit_item_timer = 0;
		#limit_item_check_count = 0;

		end;
	}


	// ========================================================
	// VERIFICAÇÃO
	// ========================================================

	if (
		func_CheckMapItems(.@map$)
	) {

		close2;

		#limit_item_timer = 0;
		#limit_item_check_count = 0;

		warp "SavePoint",0,0;

		end;
	}


	// ========================================================
	// PRIMEIRA VERIFICAÇÃO EXTRA
	// ========================================================

	#limit_item_check_count++;

	if (
		#limit_item_check_count == 1
	) {

		addtimer 500,
			.npc_name$+"::OnCheck";

		end;
	}


	// ========================================================
	// VERIFICAÇÃO CONTÍNUA
	// ========================================================

	#limit_item_check_count = 0;

	addtimer
		(.check_interval_second * 1000),
		.npc_name$+"::OnCheck";

	end;


	// ============================================================
	// LOGOUT
	// ============================================================

	OnPCLogoutEvent:

		#limit_item_timer = 0;
		#limit_item_check_count = 0;

		end;


	// ============================================================
	// @LIMITEDITEM
	// ============================================================

	OnAtCommand:

		.@map$ = .@atcmd_parameters$;

		if (
			getd(".li_item_size_"+.@map$) <= 0
		) {

			dispbottom
				"[LimitedItem] Nenhuma restrição encontrada para este mapa.";

			end;
		}


		dispbottom
			"========== LimitedItem ==========";

		dispbottom
			"Mapa: "+.@map$;

		dispbottom
			"Nível GM: "+
			getd(".li_gmlv_"+.@map$);

		.@size =
			getd(".li_item_size_"+.@map$);

		copyarray
			.@array,
			getd(".li_item_"+.@map$),
			.@size;

		.@i = 0;

		while (.@i < .@size) {

			.@itemid = .@array[.@i];
			.@max = .@array[.@i + 1];

			dispbottom
				getitemname(.@itemid)+
				" ["+.@itemid+"] - Máximo: "+
				.@max;

			.@i += 2;
		}

		dispbottom
			"================================";

		end;
}
