function	script	rouletteSystem	{

	// Animação da roleta
	for(.@ii = 1; .@ii < 100; .@ii++) {
		.@id = rand(1,17);
		cutin "roulette_"+.@id,4;
		sleep2 50;
		cutin "",255;
	}

	for(.@ii = 1; .@ii < 20; .@ii++) {
		.@id = rand(1,17);
		cutin "roulette_"+.@id,4;
		sleep2 300;
		cutin "",255;
	}

	for(.@ii = 1; .@ii < 5; .@ii++) {
		.@id = rand(1,17);
		cutin "roulette_"+.@id,4;
		sleep2 700;
		cutin "",255;
	}

	// 5% de chance de não ganhar nada
	if(rand(1,100) <= 5) {
		cutin "roulette_18",4;
		dispbottom "[ Roleta da Sorte ]: Nossa como você é azarado!";
		sleep2 3000;
		cutin "",255;
		end;
	}

	// Soma dos pesos
	.@total = 0;

	for(.@i = 0; .@i < getarraysize($@roulette_item); .@i++)
		.@total += $@roulette_chance[.@i];

	if(.@total <= 0) {
		dispbottom "[ Roleta da Sorte ]: Erro interno no sorteio.";
		end;
	}

	// Sorteia um número baseado nos pesos
	.@r = rand(1,.@total);

	.@acum = 0;

	for(.@i = 0; .@i < getarraysize($@roulette_item); .@i++) {

		.@acum += $@roulette_chance[.@i];

		if(.@r <= .@acum) {

			cutin "roulette_"+$@roulette_cutin[.@i],4;

			// Anúncio global dos itens especiais
			for(.@j = 0; .@j < getarraysize($@roulette_special); .@j++) {

				if($@roulette_item[.@i] == $@roulette_special[.@j]) {

					announce "[ Roleta da Sorte ]: Atenção Heróis! "+strcharinfo(0)+" acabou de receber o item '"+getitemname($@roulette_item[.@i])+"'!",bc_all;
					break;
				}
			}

			getitem $@roulette_item[.@i], $@roulette_qnt[.@i];

			dispbottom "[ Roleta da Sorte ]: Você recebeu "+$@roulette_qnt[.@i]+"x "+getitemname($@roulette_item[.@i])+".";

			sleep2 3000;
			cutin "",255;
			end;
		}
	}

	dispbottom "[ Roleta da Sorte ]: Erro inesperado.";
	cutin "",255;
	end;
}
