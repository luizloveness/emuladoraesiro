//===== rAthena Script =======================================
//= Reset NPC (English version, revised)
//===== Description ==========================================
//= Resets skills, stats, or both.
//= Optional: limit number of resets per character.
//===== Credits ==============================================
//= Original base by Kisuka / Euphy / Stolao / sader1992
//= Revision and improvements by ChatGPT
//============================================================
prontera,137,166,5	script	Reset Girl	124,{

	// Settings: [ Skill Reset Cost , Stat Reset Cost , Both Reset Cost , Reset Limit (0 = unlimited) ]
	setarray .@Reset, 5000, 5000, 9000, 0;

	// Ensure initialization of reset limit variable
	if (!reset_limit) set reset_limit,0;

	mes "[Reset Girl]";
	if (.@Reset[3] && reset_limit >= .@Reset[3]) {
		mes "Sorry, you can only reset "
			+ callfunc("F_InsertPlural",.@Reset[3],"time") + " in your life.";
		close;
	}

	mes "Hello, I am the Reset Girl.";
	mes "I can help you with:";
	mes "- Stats Reset: ^0055FF"+ callfunc("F_InsertComma",.@Reset[1]) +"z^000000";
	mes "- Skills Reset: ^0055FF"+ callfunc("F_InsertComma",.@Reset[0]) +"z^000000";
	mes "- Both: ^0055FF"+ callfunc("F_InsertComma",.@Reset[2]) +"z^000000";
	if (.@Reset[3]) 
		mes "You may only reset "
			+ callfunc("F_InsertPlural",.@Reset[3],"time")
			+ " in your life, so use it wisely!";
	
	next;
	set .@i, select("Reset Skills:Reset Stats:Reset Both:Cancel");

	// Cancel
	if (.@i == 4) {
		mes "[Reset Girl]";
		mes "Alright, come back if you change your mind.";
		close;
	}

	// Check Zeny
	if (Zeny < .@Reset[.@i-1]) {
		mes "[Reset Girl]";
		mes "Sorry, you don’t have enough Zeny.";
		close;
	}

	// Confirmation if reset limit exists
	if (.@Reset[3]) {
		mes "[Reset Girl]";
		mes "Remember: you may only reset "
			+ callfunc("F_InsertPlural",.@Reset[3],"time") + " in your life.";
		mes "Are you sure you want to continue?";
		if (select("Let me think...:Yes, reset me") == 1) close;
	}

	// Deduct Zeny
	set Zeny, Zeny - .@Reset[.@i-1];

	// Perform reset
	if (.@i & 1) {
		sc_end_class;
		ResetSkill;
	}
	if (.@i & 2) {
		ResetStatus;
	}

	// Increase reset counter if limit is enabled
	if (.@Reset[3]) set reset_limit, reset_limit + 1;

	mes "[Reset Girl]";
	mes "All done! Your reset is complete.";
	close;
}
