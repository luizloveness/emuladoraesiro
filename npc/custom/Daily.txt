//============================================================
// EVENT: NAGHT SIEGER KILL (PARA TODOS DA PARTY)
//============================================================
OnNPCKillEvent:

    // Se o monstro morto NÃO é o Naght Sieger, sai
    if (killedrid != 1956) goto L_LHZCheck;

    // Obtém ID da party
    .@party_id = getcharid(1);
    if (!.@party_id) { // Jogador não está em party
        if (#dq_naght_active) {
            set #dq_naght_kill, 1;
            dispbottom "^00FF00[Daily Quest] Naght Sieger defeated!^000000";
        }
        end;
    }

    // Pega membros da party
    getpartymember .@party_id, 1; // Conta
    getpartymember .@party_id, 2; // GID
    getpartymember .@party_id, 0; // Names

    // Marca a quest para todos os membros
    for (.@i = 0; .@i < $@partymembercount; .@i++) {

        // Verifica se o jogador existe no mapa
        if (!isloggedin($@partymemberaid[.@i])) continue;

        attachrid($@partymemberaid[.@i]);

        // Apenas quem tem a quest ativa recebe
        if (#dq_naght_active && #dq_naght_kill == 0) {
            set #dq_naght_kill, 1;
            dispbottom "^00FF00[Daily Quest] Naght Sieger defeated!^000000";
            dispbottom "^00FF00Return to the NPC.^000000";
        }

        detachrid;
    }

    end;

// abyss CHECK (safe, no end; inside)

A_AbyssCheck:

            goto S_SalamanderCheck;

                   #dq_Abyss_count + "/" + #dq_Abyss_goal + "^000000";

S_SalamanderCheck:
    if (#dq_salamander_active && killedrid == 1831) {

        .@map$ = strcharinfo(3);
        if (.@map$ != "thor_v02" && .@map$ != "thor_v03")
            goto S_siromaCheck;

        set #dq_salamander_count, #dq_salamander_count + 1;

        dispbottom "^FF6600[Salamander Daily] Salamander: " +
                   #dq_salamander_count + "/" + #dq_salamander_goal + "^000000";

        if (#dq_salamander_count >= #dq_salamander_goal)
            dispbottom "^00FF00[Salamander Daily] Completed! Return to the NPC.^000000";
    }

    goto S_siromaCheck;
