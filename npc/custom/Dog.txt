prontera,149,143,5	script	dog#h1-1::dog	81,{

	set .@lang, #langtype; // 0 = EN / 8 = PT

	if (.@lang == 0) {
		mes "[Healer]";
		mes "I have amazing healing powers!!";
		mes "You look like a person who needs them.";
		mes "Want some?";
		next;
		menu "Heal",-,"No thanks",CANCEL;
	} else {
		mes "[Curandeiro]";
		mes "Eu possuo incríveis poderes de cura!!";
		mes "Você parece precisar deles.";
		mes "Deseja receber minha ajuda?";
		next;
		menu "Curar",-,"Não, obrigado",CANCEL;
	}

	if (vip_status(1)) {

		// Cooldown apenas no NPC de Amatsu
		if (strnpcinfo(3) == "dog#h1-12") {
			if ((gettimetick(2) - #last_vipbuff_AMATSU) < 120) {
				mes "[Healer]";
				if (.@lang == 0)
					mes "You must wait 2 minutes before receiving VIP buffs again (Amatsu).";
				else
					mes "Você deve aguardar 2 minutos para receber os buffs VIP novamente (Amatsu).";
				close;
			}
		}

		mes "[Healer]";
		if (.@lang == 0)
			mes "You're a VIP! Full service incoming!";
		else
			mes "Você é VIP! Receba nosso serviço completo!";

		specialeffect2 EF_HEAL2;
		percentheal 100,100;

		// Buffs VIP
		sc_start SC_BLESSING,300000,10;
		sc_start SC_INCREASEAGI,300000,10;
		sc_start SC_MAGNIFICAT,300000,5;
		sc_start SC_FOOD_STR_CASH,300000,5;
		sc_start SC_FOOD_AGI_CASH,300000,5;

		// Identifica todos os itens
		identifyall;

		// Salva o cooldown após utilizar o NPC
		if (strnpcinfo(3) == "dog#h1-12")
			#last_vipbuff_AMATSU = gettimetick(2);

		#last_vipbuff = gettimetick(2);

		if (.@lang == 0) {
			mes "You're fully healed, all your items have been identified, and you received exclusive VIP buffs!";
		} else {
			mes "Você foi completamente curado, todos os seus itens foram identificados e você recebeu os buffs exclusivos de VIP!";
		}

	} else {

		heal 35000,35000;

		mes "[Healer]";
		if (.@lang == 0)
			mes "You are now healed!";
		else
			mes "Você foi curado!";
	}

	if (.@lang == 0)
		mes "Have fun adventuring!!";
	else
		mes "Boa aventura!!";

	close;

CANCEL:
	if (.@lang == 0) {
		mes "[Healer]";
		mes "Alright, come back if you need a heal.";
	} else {
		mes "[Curandeiro]";
		mes "Tudo bem. Volte quando precisar de uma cura.";
	}
	close;
}

-	script	VIPBuffResetOnDeath	-1,{

OnPCDieEvent:
	// Remove o cooldown do NPC de Amatsu
	#last_vipbuff_AMATSU = gettimetick(2) - 120;

	// Remove também o cooldown do @buffsvip (caso utilize esta variável)
	#last_vipbuff = gettimetick(2) - 120;

	end;
}
