//==============================================================
// NPC: Costume Converter (with VIP Discount)
// Autor: Luiz + ChatGPT
// Função: Converte automaticamente itens equipados em suas versões
//         costume equivalentes, exigindo Bronze Coins e Zeny.
//         VIPs (Group ID 5) pagam metade.
//
// Requisitos padrão:
//   • 15x Bronze Coin (ID 673)
//   • 20,000,000 Zeny
//
// Requisitos VIP (Group ID 5):
//   • 7x Bronze Coin (ID 673)
//   • 10,000,000 Zeny
//
//==============================================================

prontera,127,181,5	script	Costume Converter	409,{

	mes "[Costume Converter]";
	mes "Hello adventurer!";
	mes "I can convert your equipped items into their costume versions.";
	next;

	while (true) {
		switch(select("Show convertible items:Show requirements:Convert equipped items:Cancel")) {
			case 1:
				callsub L_ShowList;
				break;

			case 2:
				callsub L_Requirements;
				break;

			case 3:
				callsub L_Convert;
				break;

			case 4:
				mes "[Costume Converter]";
				mes "See you next time!";
				close;
		}
	}
	end;

//==============================================================
// Subrotina: Mostrar lista de itens convertíveis
//==============================================================
L_ShowList:
	mes "[Convertible Items]";
	mes "Here is a list of all items I can convert:";
	next;

	// Lista de conversões (equipamento -> costume)
	setarray .@equip_id[0],  2269,2270,2268,2267,2266,2218,5170,5172,2254,2255,2285,5121,5052,5075,5080,5091,2286,5001,5045,5118,2236,5107,5267,5132,5069,5074,5068,2247,2201,35005,35006,35007,5038;
	setarray .@costume_id[0],19552,20223,20091,19902,19528,19634,19585,31037,19529,19731,19797,20244,20159,19582,20454,20004,19551,19628,19662,19617,19982,19785,19639,19848,19618,19765,19621,19720,19830,3047,3049,3048,19547;

	for (.@i = 0; .@i < getarraysize(.@equip_id); ++.@i) {
		mes "- ^00FF00" + getitemname(.@equip_id[.@i]) + "^000000 to ^0000FF" + getitemname(.@costume_id[.@i]) + "^000000";
	}
	next;
	mes "Those are all the items I can work with.";
	close;


//==============================================================
// Subrotina: Mostrar requisitos
//==============================================================
L_Requirements:
	mes "[Conversion Requirements]";
	if (getgroupid() == 5) {
		mes "Since you are a ^00CCFFVIP^000000, your cost is reduced!";
		mes "- ^00880010,000,000^000000 zeny";
		mes "- ^8800007 Bronze Coins^000000";
	} else {
		mes "- ^00880020,000,000^000000 zeny";
		mes "- ^88000015 Bronze Coins^000000";
	}
	mes "- And the original item equipped.";
	next;
	mes "All refines, cards, and enchantments will be lost!";
	close;


//==============================================================
// Subrotina: Converter itens equipados
//==============================================================
L_Convert:
	mes "[Costume Converter]";
	mes "All refines, cards, and enchantments will be lost.";

	// Requisitos padrão
	.@req_coin_id = 673;
	.@req_coin_amt = 15;
	.@req_zeny = 20000000;

	// Desconto para VIP
	if (getgroupid() == 5) {
		mes "As a ^00CCFFVIP^000000, you get a 50% discount!";
		.@req_coin_amt = 7;
		.@req_zeny = 10000000;
	}

	mes "You must have ^880000" + .@req_coin_amt + " Bronze Coins^000000 and ^008800" + .@req_zeny + "^000000 zeny.";
	mes "Proceed?";
	if (select("Yes:No") == 2) {
		mes "Maybe next time!";
		close;
	}

	// Verificação dos requisitos
	if (countitem(.@req_coin_id) < .@req_coin_amt) {
		mes "You dont have enough ^880000Bronze Coins^000000.";
		close;
	}

	if (Zeny < .@req_zeny) {
		mes "You dont have enough ^008800zeny^000000.";
		close;
	}

	// Lista de conversões (equipamento -> costume)
	setarray .@equip_id[0],  2269,2270,2268,2267,2266,2218,5170,5172,2254,2255,2285,5121,5052,5075,5080,5091,2286,5001,5045,5118,2236,5107,2280,5132,5069,5074,5068,2247,2201,35005,35006,35007,5038;
	setarray .@costume_id[0],19552,20223,20091,19902,19528,19634,19585,31037,19529,19731,19797,20244,20159,19582,20454,20004,19551,19628,19662,19617,19982,19785,19639,19848,19618,19765,19621,19720,19830,3047,3049,3048,19547;
	.@converted = 0;

	for (.@i = 0; .@i < getarraysize(.@equip_id); ++.@i) {
		if (isequipped(.@equip_id[.@i])) {
			mes "Found ^00FF00" + getitemname(.@equip_id[.@i]) + "^000000 equipped.";
			mes "Converting to ^0000FF" + getitemname(.@costume_id[.@i]) + "^000000...";
			unequip .@equip_id[.@i];
			delitem .@equip_id[.@i], 1;
			getitem .@costume_id[.@i], 1;
			dispbottom "Converted: ^00FF00" + getitemname(.@equip_id[.@i]) + "^000000 to ^0000FF" + getitemname(.@costume_id[.@i]) + "^000000";
			.@converted++;
		}
	}

	if (.@converted == 0) {
		mes "You dont have any convertible items equipped.";
		close;
	}

	// Remove custos
	delitem .@req_coin_id, .@req_coin_amt;
	Zeny -= .@req_zeny;

	mes "All possible items have been converted!";
	mes "Used ^880000" + .@req_coin_amt + " Bronze Coins^000000 and ^008800" + .@req_zeny + "^000000 zeny.";
	close;


}
