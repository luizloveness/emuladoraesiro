// ================= TROCA
S_Troca:
	set .@item,getarg(0);
	set .@amount,getarg(1);
	set .@cost,getarg(2);

	set .@lang, #langtype;

	//  Sem pontos suficientes
	if(#MVP_POINTS < .@cost){
		mes "[MVP Shop]";
		if(.@lang == 0)
			mes "You don't have enough points.";
		else
			mes "Você não tem pontos suficientes.";
		close;
	}

	// CHECAGEM DE PESO
	if (!checkweight(.@item, .@amount)) {
		mes "[MVP Shop]";
		if(.@lang == 0)
			mes "You are overweight or carrying too much.";
		else
			mes "Você está com peso excessivo ou carregando itens demais.";
		close;
	}

	// EXECUTA TROCA
	set #MVP_POINTS, #MVP_POINTS - .@cost;
	getitem .@item,.@amount;

	mes "[MVP Shop]";
	if(.@lang == 0){
		mes "Exchange completed successfully!";
		mes "Remaining points: "+#MVP_POINTS;
	}else{
		mes "Troca realizada com sucesso!";
		mes "Pontos restantes: "+#MVP_POINTS;
	}

	close;


}
